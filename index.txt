function evalFormula(f,depth){
 depth=depth||0; if(depth>20) return '#CYCLE';
 if(!f) return '';
 if(f[0]!=='=') return f;
 var expr=f.slice(1);
 expr=expr.replace(/[A-Z]+[0-9]+/g, function(ref){
  var m=ref.match(/([A-Z]+)([0-9]+)/);
  if(!m) return '0';
  var colStr=m[1], rowStr=m[2];
  var cNum=0;
  for(var i=0;i<colStr.length;i++){ cNum=cNum*26+(colStr.charCodeAt(i)-64); }
  cNum=cNum-1;
  var rNum=parseInt(rowStr,10)-1;
  if(rNum>=0 && rNum<rows && cNum>=0 && cNum<cols){
   var v=data[rNum][cNum].v;
   if(data[rNum][cNum].type==='formula'){ v=evalFormula(v,depth+1); }
   var num=parseFloat(v);
   return isNaN(num)?'0':num.toString();
  }
  return '0';
 });
 try{ if(!/^[0-9+\-*\/().\s%eE]*$/.test(expr)) return '#ERR'; var res=Function('return ('+expr+')')(); return res; }catch(e){ return '#ERR'; }
}

    var arr=lines.map(function(l){ var out=[],cur='',q=false; for(var k=0;k<l.length;k++){ var ch=l[k]; if(q){ if(ch==='"'&&l[k+1]==='"'){cur+='"';k++;} else if(ch==='"') q=false; else cur+=ch; } else if(ch==='"') q=true; else if(ch===','){out.push(cur);cur='';} else cur+=ch; } out.push(cur); return out; });
    rows=Math.max(1,Math.min(100,arr.length)); cols=1; arr.forEach(function(r){ cols=Math.max(cols,r.length); }); cols=Math.min(100,cols);
    initData();
    for(var r=0;r<rows;r++) for(var c=0;c<cols;c++) data[r][c].v=(arr[r]&&arr[r][c])||'';
   }
   renderTable();
  }catch(e){ alert('ملف غير صالح / Invalid file'); }
 };
 rd.readAsText(file);
}

/* QR (canvas) - simple deterministic pattern preview */
function drawQR(cv,val){
 var ctx=cv.getContext('2d'); var n=21, cell=Math.floor(cv.width/n);
 ctx.fillStyle='#fff'; ctx.fillRect(0,0,cv.width,cv.height); ctx.fillStyle='#000';
 var h=2166136261;
 function rnd(){ h^=h<<13; h^=h>>>17; h^=h<<5; return (h>>>0)/4294967296; }
 for(var k=0;k<val.length;k++){ h=(h^val.charCodeAt(k))*16777619>>>0; }
 if(!h) h=1;
 function finder(x,y){ ctx.fillRect(x*cell,y*cell,7*cell,7*cell); ctx.fillStyle='#fff'; ctx.fillRect((x+1)*cell,(y+1)*cell,5*cell,5*cell); ctx.fillStyle='#000'; ctx.fillRect((x+2)*cell,(y+2)*cell,3*cell,3*cell); }
 for(var y=0;y<n;y++) for(var x=0;x<n;x++){ if((x<8&&y<8)||(x>n-9&&y<8)||(x<8&&y>n-9)) continue; if(rnd()>0.5) ctx.fillRect(x*cell,y*cell,cell,cell); }
 finder(0,0); finder(n-7,0); finder(0,n-7);
}
